cmake_minimum_required(VERSION 3.21)
project(razer_mixer_linux_driver VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick QuickControls2 QuickDialogs2)
if(Qt6_VERSION VERSION_GREATER_EQUAL "6.8")
    qt_policy(SET QTP0004 NEW)
endif()
add_library(mixer_core STATIC src/Config.cpp src/Audio.cpp src/Protocol.cpp src/Hid.cpp src/Control.cpp)
target_include_directories(mixer_core PUBLIC src)
target_link_libraries(mixer_core PUBLIC Qt6::Core)
target_compile_options(mixer_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(razer-mixer-driver src/main.cpp src/Service.cpp)
target_link_libraries(razer-mixer-driver PRIVATE mixer_core)
qt_add_executable(razer-mixer app/main.cpp app/Controller.cpp)
target_link_libraries(razer-mixer PRIVATE mixer_core Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2)
set_source_files_properties(app/qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(razer-mixer URI RazerMixer VERSION 1.0 RESOURCE_PREFIX "/qt/qml"
    QML_FILES app/qml/Main.qml app/qml/Theme.qml app/qml/ActionButton.qml
              app/qml/Channel.qml app/qml/Lighting.qml app/qml/ColorField.qml app/qml/Routing.qml)
install(TARGETS razer-mixer razer-mixer-driver RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/razer-mixer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/razer-mixer.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
configure_file(packaging/razer-mixer-bridge.service.in razer-mixer-bridge.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/razer-mixer-bridge.service DESTINATION ${CMAKE_INSTALL_DATADIR}/razer-mixer)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(mixer-tests tests/test_core.cpp)
    target_link_libraries(mixer-tests PRIVATE mixer_core Qt6::Test)
    add_test(NAME mixer-core COMMAND mixer-tests)
endif()
